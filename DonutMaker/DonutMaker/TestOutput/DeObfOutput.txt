using System;
using System.Threading;
namespace SimpleConsoleApp
{
    internal class Program
    {
        static void Main(string [] args){
            Console.WriteLine("=== Simple Console Demo ===");
            Console. WriteLine("Type anything and I'll animate it.");
            Console . WriteLine("Type 'exit' to quit.");
            Console . WriteLine();
            while(true){
                Console .Write("Input: ");
                string? input = Console .ReadLine();
                if(input == null) continue;
                if(input . Equals("exit", StringComparison .OrdinalIgnoreCase)) break;
                Console. WriteLine("\nAnimating forward...");
                TypeWriter(input, 40);
                string reversed = Reverse(input);
                Console .WriteLine("\nAnimating backward...");
                TypeWriter(reversed, 40);
                Console . WriteLine("\n");}
                Console . WriteLine("Goodbye!");}
                static void TypeWriter(string text, int delayMs){
                    foreach(char c in text){
                        Console .Write(c);
                        Thread . Sleep(delayMs);}
                        Console.WriteLine();}
                        static string Reverse(string s){
                            char [] arr = new char [ s . Length];
                            int j = 0;
                            for(int i = s . Length - 1;
                            i >= 0;
                            i --){
                                arr[ j++] = s [ i];}
                                return new string(arr);}}}
